function parseLedger(markdown) {
  // The archive includes a decoy "Transaction Ledger (working copy)".
  // Authoritative ledger begins at the exact heading "## Transaction Ledger".
  const heading = "## Transaction Ledger
";
  let start = markdown.indexOf(heading);
  if (start < 0) return [];

  const working = markdown.indexOf("## Transaction Ledger (working copy)");
  if (working >= 0 && start === working) {
    start = markdown.indexOf(heading, working + 1);
    if (start < 0) return [];

  const slice = markdown.slice(start);
  const rows = [];
  const blocks = slice.split(/^### /m).slice(1);

  for (const block of blocks) {
    const lines = block.split("
");
    const transactionId = lines[0].trim();
    if (!transactionId.startsWith("TXN-")) continue;

    const row = { transaction_id: transactionId };
    let provisional = false;

    for (const line of lines) {
      const trimmed = line.trim();
      if (trimmed.startsWith("amount:")) {
        row.amount = Number(trimmed.split(":")[1].trim());
      } else if (trimmed.startsWith("owner:")) {
        row.owner = trimmed.split(":")[1].trim();
      } else if (trimmed.startsWith("status:")) {
        row.status = trimmed.split(":")[1].trim();
      } else if (trimmed.startsWith("date:")) {
        row.date = trimmed.split(":")[1].trim();
      } else if (trimmed.startsWith("provisional:")) {
        provisional = trimmed.split(":")[1].trim().toLowerCase() === "true";
      }
    }

    if (provisional) continue;

    const existingIdx = rows.findIndex((r) => r.transaction_id === transactionId);
    if (existingIdx >= 0) rows.splice(existingIdx, 1);
    rows.push(row);
  }

  return rows;
}

module.exports = { parseLedger };
