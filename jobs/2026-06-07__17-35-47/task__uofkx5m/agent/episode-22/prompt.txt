> // Utility to slice a markdown document between exact heading lines.
> // Brief 01 requires locating section boundaries with regex anchored to whole
> // heading lines at line start.
>
> function escapeRegex(s) {
>   return s.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
> }
>
> function findHeadingIndex(markdown, headingLine) {
>   const re = new RegExp(`^${escapeRegex(headingLine)}$`, "m");
>   const match = re.exec(markdown);
>   return match ? match.index : -1;
> }
>
> function sliceSection(markdown, startHeadingLine, endHeadingLine) {
>   const start = findHeadingIndex(markdown, startHeadingLine);
>   if (start < 0) return "";
>   if (!endHeadingLine) return markdown.slice(start);
>   const end = findHeadingIndex(markdown, endHeadingLine);
>   if (end < 0 || end <= start) return markdown.slice(start);
>   return markdown.slice(start, end);
> }
>
> module.exports = { sliceSection, findHeadingIndex };
